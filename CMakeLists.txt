function(configure_native_target target)
  target_compile_features(${target} PRIVATE cxx_std_20)
  target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX VK_NO_PROTOTYPES VK_USE_PLATFORM_WIN32_KHR)
  target_include_directories(${target} SYSTEM PRIVATE third_party/Vulkan-Headers/include third_party/xess/inc)
  target_link_libraries(${target} PRIVATE d3d12 dxgi dxguid)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8 /EHsc)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wno-missing-field-initializers)
    target_link_options(${target} PRIVATE -static)
  endif()
endfunction()

# Product dependencies receive one build mode, including object libraries.
# Standalone fixtures explicitly retain their development/release test profile.
add_library(xess-build-mode INTERFACE)
target_compile_definitions(xess-build-mode INTERFACE XESS_RELEASE_MINIMAL=$<BOOL:${XESS_RELEASE_MINIMAL}>)
add_library(xess-fixture-build-mode INTERFACE)
target_compile_definitions(xess-fixture-build-mode INTERFACE XESS_RELEASE_MINIMAL=0)
add_library(xess-release-fixture-build-mode INTERFACE)
target_compile_definitions(xess-release-fixture-build-mode INTERFACE XESS_RELEASE_MINIMAL=1)
set(xess_product_targets)
set(xess_pending_targets xess-vulkan-observer EndfieldXeSSLauncher)
while(xess_pending_targets)
  list(POP_FRONT xess_pending_targets current)
  if(NOT TARGET ${current} OR current IN_LIST xess_product_targets)
    continue()
  endif()
  list(APPEND xess_product_targets ${current})
  get_target_property(links ${current} LINK_LIBRARIES)
  get_target_property(sources ${current} SOURCES)
  foreach(dependency IN LISTS links sources)
    if(dependency MATCHES "^\\$<(LINK_ONLY|TARGET_OBJECTS):([^>]+)>$")
      set(dependency "${CMAKE_MATCH_2}")
    endif()
    if(TARGET ${dependency})
      list(APPEND xess_pending_targets ${dependency})
    endif()
  endforeach()
endwhile()
get_property(xess_all_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
set(xess_flavor_audit "Product build flavor: ${XESS_RELEASE_MINIMAL}\n")
foreach(current IN LISTS xess_all_targets)
  get_target_property(kind ${current} TYPE)
  if(kind STREQUAL "INTERFACE_LIBRARY")
    continue()
  endif()
  get_target_property(definitions ${current} COMPILE_DEFINITIONS)
  foreach(definition IN LISTS definitions)
    if(definition MATCHES "^XESS_RELEASE_MINIMAL")
      message(FATAL_ERROR "${current}: local build-flavor override is forbidden; use the common interface target")
    endif()
  endforeach()
  if(current IN_LIST xess_product_targets)
    target_link_libraries(${current} PRIVATE xess-build-mode)
    string(APPEND xess_flavor_audit "${current} -> xess-build-mode\n")
  elseif(current STREQUAL "xess-release-profile-test")
    target_link_libraries(${current} PRIVATE xess-release-fixture-build-mode)
  else()
    target_link_libraries(${current} PRIVATE xess-fixture-build-mode)
  endif()
endforeach()
file(WRITE "${CMAKE_BINARY_DIR}/build-flavor-targets.txt" "${xess_flavor_audit}")
add_test(NAME build-flavor-contract COMMAND ${CMAKE_COMMAND}
  -DCXX=${CMAKE_CXX_COMPILER} -DSOURCE_ROOT=${CMAKE_CURRENT_SOURCE_DIR}
  -DOUTPUT_ROOT=${CMAKE_BINARY_DIR} -P ${CMAKE_CURRENT_SOURCE_DIR}/scripts/test-build-flavor.cmake)
